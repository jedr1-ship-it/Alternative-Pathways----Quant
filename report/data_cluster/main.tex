% ================= Opening =================
\begin{frame}{Motivation and problem}
  \vspace{0.8em}
  {\color{navy}\bfseries Motivation}\\[0.5em]
  \begin{itemize}\setlength\itemsep{0.6em}
    \item Understand why teachers leave the profession.
    \item Is there a real policy margin to soften teacher shortages by
          retaining teachers --- or bringing them back?
  \end{itemize}
  \vspace{1.2em}
  {\color{navy}\bfseries Problem}\\[0.5em]
  \begin{itemize}\setlength\itemsep{0.6em}
    \item Official microdata on teachers are not publicly available for
          many countries.
    \item We start with the United States: the country with the most
          information available.
  \end{itemize}
\end{frame}

\begin{frame}{Solution}
  \vspace{1.2em}
  Use the Current Population Survey to
  \vspace{0.8em}
  \begin{enumerate}\setlength\itemsep{1.2em}
    \item[A.] reproduce and extend the official statistics on teachers
          leaving the profession;
    \item[B.] having achieved A, study in detail \emph{how many} leave,
          \emph{where} they go, and \emph{why}.
  \end{enumerate}
\end{frame}

% ================= CPS =================
\begin{frame}[label=maincps]{The Current Population Survey}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{1.0em}
    \item The monthly household survey of the United States, run by the
          Census Bureau for the Bureau of Labor Statistics. Nationally
          representative of the civilian non-institutional population.
    \item We pool 28 consecutive March supplements, 1998--2025:
          \textbf{5.0 million} person-year records.
    \item \textbf{104,545} of them are teachers --- about 3,700 per year.
  \end{itemize}
  \vfill
  \appbtn{appunemp}{unemployment}\;
  \appbtn{approtation}{interviews}\;
  \appbtn{appmigration}{migration}\;
  \appbtn{appidentity}{identity}
\end{frame}

% ================= March supplement =================
\begin{frame}{The March supplement (ASEC)}
  \vspace{0.8em}
  \begin{itemize}\setlength\itemsep{1.0em}
    \item Run by the Census Bureau each March since 1948: the source of
          the official annual income, poverty and health-insurance
          statistics.
    \item About 90,000 households: the March sample plus oversamples
          interviewed in February and April.
    \item Every adult reports on the \emph{previous calendar year}:
          longest job held, weeks worked, earnings, benefits.
  \end{itemize}
\end{frame}

      {\color{navy}\bfseries B. Ask about last year}\\[0.7em]

      {\small Taught as last year's longest job, not teaching this March
       $\Rightarrow$ left.}
    \end{column}

% ================= Disadvantages =================
\begin{frame}[label=maindisadv]{Disadvantages of each option}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      {\color{navy}\bfseries A. Following individuals}\\[0.6em]
      {\small
      \begin{enumerate}\setlength\itemsep{0.5em}
        \item Identity is not certain: validation can admit noisy people.
        \item Movers are lost: $\sim$25\% of addresses are not found a
              year later.
        \item Normal household attrition on top.
        \item Strong seasonality in teaching.
        \item Many ways to define the rate (next slide).
      \end{enumerate}}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\color{navy}\bfseries B. Retrospective March}\\[0.6em]
      {\small
      \begin{enumerate}\setlength\itemsep{0.5em}
        \item Recall bias.
        \item Less sample: one observation per person per year
              ($\sim$3,700 teachers) against $\sim$11,000 monthly
              teacher observations.
        \item Only March-to-March transitions; within-year spells are
              smoothed away.
      \end{enumerate}}
    \end{column}
  \end{columns}
  \vfill
  \appbtn{applink}{linkage loss}\;
  \appbtn{appgradient}{attachment}\;
  \appbtn{appidentity}{validation}
\end{frame}

\begin{frame}{What this says for policy}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{1.1em}
    \item Of every 100 leavers, about 36 retire: a margin for
          \emph{delaying} exit, not for return.
    \item 29 move to another job --- half of them to destinations that
          do not require a degree.
    \item 11 are unemployed and 24 leave the labor force before 55;
          about 13\% of the latter are studying.
    \item The retrievable margin is real but bounded: policy competes
          with retirement, family time and re-training --- not only with
          other employers.
  \end{itemize}
\end{frame}

    \;+\; \mathbf{a}_{it}'\boldsymbol\beta_1
    \;+\; \mathbf{c}_{it}'\boldsymbol\beta_2
    \;+\; \mathbf{f}_{it}'\boldsymbol\beta_3
    \;+\; \mathbf{h}_{it}'\boldsymbol\beta_4
    \;+\; \mathbf{d}_{it}'\boldsymbol\beta_5
    \;+\; \delta_t \;+\; \varepsilon_{it}

  {\small Probit; average marginal effects reported. 2018--2024,

% ================= Conclusions =================
\begin{frame}{Conclusions}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{1.0em}
    \item A household survey can reproduce the official picture of
          teacher attrition --- and extend it yearly, with covariates
          records do not have.
    \item Who leaves: the weakly attached, the low-paid, those without
          pension coverage, new mothers, teachers with a disability.
    \item We observe only part of the motives --- financial and
          life-cycle, not working conditions --- so these are margins,
          not the full story.
    \item What could bring teachers back: flexibility, economic
          security, pension coverage, and re-entry paths for those who
          left to study or care.
  \end{itemize}
\end{frame}

\begin{frame}[label=appunemp]{How the CPS measures unemployment}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{0.9em}
    \item The \emph{basic monthly} interview measures labor force status
          in the reference week --- that alone produces the official
          monthly unemployment rate. Not the March supplement.
    \item The rest of the survey: monthly labor statistics (employment,
          hours, earnings) and topical supplements (school enrollment in
          October, voting in November, tobacco, computer use).
    \item The March supplement adds the annual income block: the source
          of income, poverty and health-insurance statistics --- and of
          our retrospective measure.
  \end{itemize}
  \backbtn{maincps}
\end{frame}

\begin{frame}[label=approtation]{Are interviews monthly? Who is new each month?}
  \vspace{0.3em}
  {\small Yes: every month, one eighth of the sample sits in each
   position of the 4-8-4 rotation. Verified in the microdata:}\\[0.7em]
  \begin{center}
  \small
  \begin{tabular}{lcccccccc}
    \toprule
    Month in sample     & 1 & 2 & 3 & 4 & 5 & 6 & 7 & 8 \\
    \midrule
    March 2023 (\%)     & 12.5 & 12.9 & 12.8 & 12.5 & 12.2 & 12.4 & 12.3 & 12.3 \\
    July 2023 (\%)      & 12.3 & 12.9 & 12.9 & 12.5 & 12.2 & 12.2 & 12.5 & 12.4 \\
    January 2024 (\%)   & 12.4 & 12.9 & 12.9 & 13.1 & 11.9 & 12.0 & 12.1 & 12.6 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.5em}
  {\small So each month: $\sim$12.5\% new households, $\sim$37.5\% in
   months 2--4, $\sim$50\% returning in months 5--8. Month-to-month
   overlap 75\%; year-to-year overlap 50\%, by design.}
  \backbtn{maincps}
\end{frame}

\begin{frame}[label=appmigration]{Internal migration in the United States}
  \vspace{0.8em}
  \begin{itemize}\setlength\itemsep{1.0em}
    \item In 2023, \textbf{7.8\%} of Americans changed residence within
          the year --- 25.6 million people, the lowest rate on record.
    \item 3.2\% moved to a different county; 1.4\% to a different state.
    \item This is the loss channel of any address-based panel: linkage
          follows dwellings, not people.
  \end{itemize}
  \vspace{0.6em}
  {\footnotesize\color{mut} Census Bureau, CPS ASEC, historical table
   A-1 (1948--2023).}
  \backbtn{maincps}
\end{frame}

\begin{frame}[label=appidentity]{Is the same person validated year to year?}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{0.9em}
    \item In the field, yes: the interviewer carries the household roster
          forward, each member keeps a line number, and occupation uses
          dependent interviewing between consecutive months.
    \item The \emph{public} files carry no person identifier. We
          reconstruct the roster: household id + dwelling id + line
          number, validated on sex, race, and age advancing 0--2 years.
    \item Validation is not watertight: a mover replaced by someone of
          the same sex and similar age can slip through.
  \end{itemize}
  \backbtn{maindisadv}
\end{frame}

\begin{frame}[label=applink]{How much is lost in linking, and who}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{0.9em}
    \item Of the addresses due to return twelve months later,
          \textbf{74.5\%} are found again (March 2023 $\to$ March 2024);
          the missing quarter is moves plus household non-response.
    \item Teachers link \emph{more} than others, not less: $\sim$80\% of
          teacher observations vs $\sim$76\% of other college graduates.
    \item The loss is still selective: movers leave teaching more, so
          panel measures are lower bounds on mobility-related exit.
  \end{itemize}
  \backbtn{maindisadv}
\end{frame}

\begin{frame}[label=appcodes]{Is the teacher group constant across code changes?}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{1.0em}
    \item The five-code K--12 block maps one-to-one across the 1990,
          2002/2010 and 2018 classifications; we always use the block,
          never a single code across regimes.
    \item Evidence: re-computing the whole series on the narrow core
          (dropping special education and other teachers) moves the
          pooled rate from 8.67 to 8.57 --- a 0.1 point change.
    \item The 2003 and 2020 reclassifications leave no visible break in
          the series.
  \end{itemize}
  \backbtn{maincodes}
