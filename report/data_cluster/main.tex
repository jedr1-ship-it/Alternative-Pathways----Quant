% ================= Why =================
\begin{frame}{Why this project}
  \vspace{1.0em}
  \begin{itemize}\setlength\itemsep{1.5em}
    \item We study the decision to leave teaching. That needs each person's
          leave-or-stay outcome, plus covariates.
    \item Administrative records count best --- but access is hard, coverage
          is partial, covariates are few.
    \item So we build the outcome from a household survey. This talk is
          about doing that carefully.
  \end{itemize}
\end{frame}

% ================= Contribution =================
\begin{frame}{Contribution}
  \vspace{1.0em}
  \begin{enumerate}\setlength\itemsep{1.5em}
    \item \msg{A user's guide} to building a leaving rate from a survey:
          each choice, each error, each effect on the number.
    \item \msg{An implementation} on 28 years of the CPS, checked against
          the NCES reference.
    \item \msg{The substance:} who leaves, and what correlates with it.
  \end{enumerate}
\end{frame}

\divframe{1}{The data}

% ================= Official statistics =================
\begin{frame}{Where do the official numbers come from?}
  \vspace{1.0em}
  \begin{itemize}\setlength\itemsep{1.5em}
    \item National reference: the NCES Teacher Follow-up Survey. Itself a
          survey. Eight waves since 1988.
    \item Administrative data: state by state, public sector, and they
          measure system exit, not profession exit.
    \item No national administrative count exists.
  \end{itemize}
\end{frame}

      \end{tabular}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \msg{Ask about last year}\\[0.8em]
      {\small Each March, the supplement asks about the previous calendar
       year: longest job, weeks, earnings.}\\[0.6em]
      {\small Taught last year, not teaching now $\Rightarrow$ left.}\\[0.6em]
      {\small\color{mut} The measure used in the literature.}
    \end{column}
  \end{columns}
\end{frame}

% ================= Four definitions =================
\begin{frame}{Four definitions, one language}
  \vspace{0.2em}
  Four interviews in year 1, four in year 2, twelve months apart.\\[1.0em]
  \begin{tabular}{p{3.0cm}p{4.3cm}p{4.3cm}}
    \toprule
                        & A teacher:                & A leaver:                 \\
    \midrule
    Any sighting        & seen teaching in $\geq$ 1 of the 4 year-1 interviews
                        & seen teaching in none of the 4 year-2 interviews     \\[1.1em]
    Month pairs         & each teaching interview counts on its own
                        & not seen teaching twelve months later                \\[1.1em]
    \msg{Panel (benchmark)} & seen teaching in $\geq$ 2 of the 4 year-1 interviews
                        & seen teaching in none of the 4 year-2 interviews$^{*}$ \\[1.1em]
    Annual recall       & teaching was the longest job of last year
                        & not teaching at the March interview                  \\
    \bottomrule
  \end{tabular}\\[0.8em]
  {\footnotesize\color{mut} $^{*}$At least two year-2 interviews, one outside
   the summer. Annual recall is the only one not built by counting
   interviews.}
\end{frame}

% ================= Why lines differ =================
\begin{frame}{Why the lines differ}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{1.3em}
    \item[\msg{18.2}] \msg{Any sighting.} Lets in substitutes and one-month
          spells.
    \item[\msg{15.4}] \msg{Month pairs.} Counts the same person several
          times.
    \item[\msg{13.0}] \msg{Panel.} One verdict per person --- but one leaver
          in six is back within three months.
    \item[\msg{8.6}] \msg{Annual recall.} The official concept: the year's
          main job.
    \item[\msg{8.0}] \msg{NCES follow-up.} The reference.
  \end{itemize}
  \vspace{0.6em}
  {\footnotesize\color{mut} The choice, not the data, drives the spread.}
\end{frame}

% ================= Five messages =================
\begin{frame}{Five messages}
  \vspace{0.6em}
  \begin{enumerate}\setlength\itemsep{1.4em}
    \item Economics beats demographics.
    \item Exits concentrate among part-time, part-year, low-paid teachers.
    \item Those exits leave employment --- not for better jobs.
    \item A birth is the largest single trigger.
    \item Pensions retain, and states differ widely.
  \end{enumerate}
\end{frame}

% ================= Equation =================
\begin{frame}{The econometric specification}
  \vspace{0.2em}
  \[
    \Pr\bigl(\text{Leave}_{it}=1 \mid x_{it}\bigr)
    \;=\;
    \Phi\Bigl(
      \underbrace{\text{attach}_{it}'\beta_1}_{\substack{\text{full year,}\\ \text{part-time}}}
      +\underbrace{\text{comp}_{it}'\beta_2}_{\substack{\text{pension,}\\ \text{public school}}}
      +\underbrace{\text{fam}_{it}'\beta_3}_{\substack{\text{new baby, children,}\\ \text{marital status}}}
      +\underbrace{\text{dem}_{it}'\beta_4}_{\substack{\text{age, age}^2\text{, sex,}\\ \text{race, education}}}
      +\;\delta_t
    \Bigr)
  \]
  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{0.5em}
    \item {\small Probit, 2015--2024, year effects, $n=28{,}900$. AMEs in
          points.}
    \item {\small Robust: logit, LPM, state and year FE, full window,
          occupation-pair definition.}
  \end{itemize}
\end{frame}

% ================= Conclusion =================
\begin{frame}{Conclusion}
  \vspace{1.0em}
  \begin{itemize}\setlength\itemsep{1.6em}
    \item A labour force survey measures attrition yearly --- and matches
          the official picture.
    \item The definition is first-order: 5 to 19 percent from the same
          records.
    \item Attachment, pay, pensions and births drive exits. And the design
          travels to any LFS.
  \end{itemize}
\end{frame}
