% ================= CPS =================
\begin{frame}[label=maincps]{The Current Population Survey}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{1.0em}
    \item The monthly household survey of the United States, run by the
          Census Bureau for the Bureau of Labor Statistics, and nationally
          representative of the civilian non-institutional population.
    \item We pool 28 consecutive March supplements, 1998--2025:
          \textbf{5.0 million} person-year records.
    \item Among them, \textbf{104,545} are teachers, about 3,700 per year.
  \end{itemize}

% ================= March supplement =================
\begin{frame}{The March supplement (ASEC)}
  \vspace{0.8em}
  \begin{itemize}\setlength\itemsep{1.0em}
    \item Run by the Census Bureau each March since 1948: the source of
          the official annual income, poverty and health-insurance
          statistics.
    \item About 90,000 households: the March sample plus oversamples
          interviewed in February and April.
    \item Every adult reports on the previous year their longest job
          held, weeks worked, earnings and benefits.
  \end{itemize}
\end{frame}

% ================= Identifying a teacher =================
\begin{frame}[label=maincodes]{Who counts as a teacher?}
  \vspace{0.6em}
  \begin{itemize}
    \item The census assigns every worker a detailed occupation code for
      the job held longest last year. We count as a teacher anyone
      classified in one of the five school-teaching categories:
      \begin{itemize}
        \item[\textbf{\color{navy}(a)}] Preschool and kindergarten teachers
        \item[\textbf{\color{navy}(b)}] Elementary and middle school teachers
        \item[\textbf{\color{navy}(c)}] Secondary school teachers
        \item[\textbf{\color{navy}(d)}] Special education teachers
        \item[\textbf{\color{navy}(e)}] Other teachers and instructors
      \end{itemize}
    \item[]
    \item Our results are robust to this boundary: dropping special
      education teachers (d) and the residual category (e) moves the
      pooled leaving rate by 0.1 points (8.7 to 8.6).
  \end{itemize}
  \vfill
  \appbtn{appcodes}{traceability}
\end{frame}

% ================= Disadvantages =================
\begin{frame}[label=maindisadv]{Disadvantages of each option}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      {\color{navy}\bfseries A. Following individuals}\\[0.6em]
      {\small
      \begin{enumerate}\setlength\itemsep{0.5em}
        \item Identity is not certain: validation can admit noisy people.
        \item Movers are lost: $\sim$25\% of addresses are not found a
              year later.
        \item Normal household attrition on top.
        \item Strong seasonality in teaching.
        \item Many ways to define the rate (next slide).
      \end{enumerate}}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\color{navy}\bfseries B. Retrospective March}\\[0.6em]
      {\small
      \begin{enumerate}\setlength\itemsep{0.5em}
        \item Recall bias.
        \item Less sample: one observation per person per year
              ($\sim$3,700 teachers) against $\sim$11,000 monthly
              teacher observations.
        \item Only March-to-March transitions; within-year spells are
              smoothed away.
      \end{enumerate}}
    \end{column}
  \end{columns}
  \vfill
  \appbtn{applink}{linkage loss}\;
  \appbtn{appgradient}{definition sensitivity}\;
  \appbtn{appidentity}{validation}

\begin{frame}{What this says for policy: three margins, quantified}
  \vspace{0.4em}
  \begin{itemize}\setlength\itemsep{1.0em}
    \item {\color{navy}\bfseries Retirees.} About 3\% of teachers
          retire each year, a third of all exits. Delaying retirement is
          a \emph{short-term} margin: it extends careers, it does not
          reintegrate anyone.
    \item {\color{navy}\bfseries Switchers.} About 2\% of teachers
          move to another job each year, more than a third of them to jobs that do
          not require a degree. A real margin, but a moderate one:
          shortages cannot be closed through retention alone.
    \item {\color{navy}\bfseries Early exits.} About 2\% of teachers
          leave the labor force before 55 each year. Mostly mothers at
          home, for whom \emph{flexibility} may be the missing piece;
          the young and childless leave to study, and need re-entry
          paths.
  \end{itemize}
\end{frame}

% ================= Conclusions =================
\begin{frame}[label=mainconcl]{Conclusions: what we learn}
  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{1.1em}
    \item Job security is fundamental to retain a teacher, our analysis
          shows that a full-year contract and pension coverage are
          strongly correlated with a lower teacher leaving rate.
    \item Leaving is concentrated at the bottom of the income
          distribution, and those who switch jobs do not necessarily
          earn more afterwards. Pay, alongside working conditions,
          still has room as a retention lever in the US.
    \item The probability of leaving is highest at the start of the
          career, making mentoring and accompaniment a priority for
          retention policies.
  \end{itemize}
  \vfill
  \appbtn{appswitchpay}{do switchers earn more?}
\end{frame}

\begin{frame}{Conclusions: where the policy margins are}
  \vspace{0.4em}
  \begin{itemize}\setlength\itemsep{0.9em}
    \item {\color{navy}\bfseries Short term.} Retirees, about 3\% of
          teachers each year, are a pool that can cover local shortages:
          experienced people, with positive spillovers on the schools
          that re-engage them.
    \item {\color{navy}\bfseries Medium term.} The young: the
          probability of leaving is highest at the start of the career.
          Mentoring and accompaniment belong where the risk is.
    \item {\color{navy}\bfseries Across the board.} Economic security
          (full-year contracts, pension coverage) and flexibility, so
          that teachers who want to stay, or to return, can; insecurity
          should not be the reason a position is left.
    \item {\color{navy}\bfseries Caveat.} We observe financial and
          life-cycle motives, not conditions inside schools: these are
          margins, not the full story.
  \end{itemize}
\end{frame}

    \end{column}
    \begin{column}{0.50\textwidth}
      {\footnotesize {\color{navy}\bfseries Why the columns differ}\\[0.4em]
      \begin{enumerate}\setlength\itemsep{0.45em}
        \item \textbf{Who counts as a teacher.} TFS follows regular
              classroom teachers of public schools, sampled in schools.
              We cover public and private, and anyone whose longest job
              was teaching, including part-year and part-time.
        \item \textbf{What counts as working.} 39 of the TFS's 57
              employed leavers hold \emph{non-teaching K--12 jobs}:
              they left the classroom, not the school system. Our
              28.8 means leaving the teaching occupation.
        \item \textbf{Who is reachable.} A school-based sample
              under-represents weakly attached teachers, exactly those
              who exit to non-participation; a household survey
              catches them.
        \item \textbf{The window.} Our column averages five years,
              2020--2024; the TFS is the single year 2021--22.
      \end{enumerate}}
    \end{column}
